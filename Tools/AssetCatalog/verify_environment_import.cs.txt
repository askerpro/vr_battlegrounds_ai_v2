// Тело метода для Unity MCP execute_code; не добавлять в сборку игрового проекта.
// Проверяет только пути из импортного манифеста, без изменения ассетов и сцен.
var plan = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Temp/LevelDesign/EnvironmentImport/import-plan.json"));
var policy = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText("Tools/AssetCatalog/catalog-policy.json"));
var issues = new System.Collections.Generic.List<string>();
var shaders = new System.Collections.Generic.Dictionary<string, int>();
int checkedAssets = 0, prefabs = 0, materials = 0, models = 0;
foreach (var item in plan["files"])
{
    string path = (string)item["path"];
    bool excluded = false;
    foreach (var root in policy["excludedAssetRoots"])
        if (path.StartsWith((string)root, System.StringComparison.OrdinalIgnoreCase)) excluded = true;
    if (excluded) continue;
    string resolved = UnityEditor.AssetDatabase.GUIDToAssetPath((string)item["guid"]);
    if (!string.Equals(path, resolved, System.StringComparison.OrdinalIgnoreCase))
        issues.Add("GUID: " + path + " -> " + resolved);
    var asset = UnityEditor.AssetDatabase.LoadMainAssetAtPath(path);
    if (asset == null)
        issues.Add("Не загружен: " + path);
    checkedAssets++;
    if (path.EndsWith(".prefab", System.StringComparison.OrdinalIgnoreCase))
    {
        prefabs++;
        var root = asset as UnityEngine.GameObject;
        if (root == null) continue;
        foreach (var filter in root.GetComponentsInChildren<UnityEngine.MeshFilter>(true))
            if (filter.sharedMesh == null) issues.Add("Нет меша: " + path + "/" + filter.name);
        foreach (var renderer in root.GetComponentsInChildren<UnityEngine.Renderer>(true))
            foreach (var material in renderer.sharedMaterials)
            {
                if (material == null) issues.Add("Нет материала: " + path + "/" + renderer.name);
                else if (material.shader == null || material.shader.name == "Hidden/InternalErrorShader")
                    issues.Add("Нет шейдера материала префаба: " + path + "/" + renderer.name);
            }
    }
    if (path.EndsWith(".fbx", System.StringComparison.OrdinalIgnoreCase)) models++;
    var mat = asset as UnityEngine.Material;
    if (mat != null)
    {
        materials++;
        string name = mat.shader == null ? "<missing>" : mat.shader.name;
        if (!shaders.ContainsKey(name)) shaders[name] = 0;
        shaders[name]++;
        if (mat.shader == null || name == "Hidden/InternalErrorShader")
            issues.Add("Нет шейдера: " + path);
        else if (UnityEditor.ShaderUtil.ShaderHasError(mat.shader))
            issues.Add("Ошибка шейдера: " + path + " / " + name);
    }
}
var report = new { checkedAssets, prefabs, materials, models, shaders, issues };
System.IO.File.WriteAllText("Temp/LevelDesign/EnvironmentImport/unity-verification.json", Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented));
return new { checkedAssets, prefabs, materials, models, shaders, issueCount = issues.Count };
