var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
System.Func<System.Type,string,System.Reflection.FieldInfo> field = (type,name) =>
{
    for (var current = type; current != null; current = current.BaseType)
    {
        var found = current.GetField(name, flags | System.Reflection.BindingFlags.DeclaredOnly); if (found != null) return found;
    }
    throw new System.MissingFieldException(type.FullName,name);
};

var savedCollections = new System.Collections.Generic.List<(object collection, object[] entries)>();
var savedDelegates = new System.Collections.Generic.List<(System.Reflection.FieldInfo field, object value)>();
var savedObjects = new System.Collections.Generic.HashSet<object>();
var savedTypes = new System.Collections.Generic.HashSet<System.Type>();
System.Action<System.Type> snapshotType = type =>
{
    if (!savedTypes.Add(type)) return;
    foreach (var member in type.GetFields(flags | System.Reflection.BindingFlags.DeclaredOnly))
    {
        if (!member.IsStatic) continue;
        object value = member.GetValue(null);
        if (typeof(System.Delegate).IsAssignableFrom(member.FieldType))
        { savedDelegates.Add((member,value)); member.SetValue(null,null); continue; }
        if (value == null || !savedObjects.Add(value)) continue;
        bool collection = value is System.Collections.IDictionary || value is System.Collections.IList ||
            (member.FieldType.IsGenericType && member.FieldType.GetGenericTypeDefinition() == typeof(System.Collections.Generic.HashSet<>));
        if (!collection) continue;
        var entries = new System.Collections.Generic.List<object>();
        if (value is System.Collections.IDictionary dictionary) foreach (System.Collections.DictionaryEntry entry in dictionary) entries.Add(entry);
        else foreach (object entry in (System.Collections.IEnumerable)value) entries.Add(entry);
        savedCollections.Add((value,entries.ToArray()));
    }
};
System.Action<System.Type> snapshotHierarchy = type => { for (var current = type; current != null && current.Namespace != null && current.Namespace.StartsWith("UltimateXR"); current = current.BaseType) snapshotType(current); };
