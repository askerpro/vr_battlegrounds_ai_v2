System.Func<UnityEngine.GameObject,UnityEngine.Matrix4x4,UnityEngine.Bounds> fitBounds=(prefab,matrix)=>{
    bool found=false;var result=new UnityEngine.Bounds();
    foreach(var mesh in prefab.GetComponentsInChildren<UnityEngine.MeshFilter>(true)) {
        var renderer=mesh.GetComponent<UnityEngine.MeshRenderer>();if(mesh.sharedMesh==null || renderer==null || !renderer.enabled)continue;
        bool visible=true;for(var t=mesh.transform;t!=null;t=t.parent) {
            if(!t.gameObject.activeSelf || t.name=="GrabHighlight" || t.name=="__ItemPreview__" || t.name=="__MagPreview__")visible=false;
            if(t==prefab.transform)break;
        }
        if(!visible)continue;
        var projection=matrix*prefab.transform.worldToLocalMatrix*mesh.transform.localToWorldMatrix;
        foreach(var vertex in mesh.sharedMesh.vertices) {var p=projection.MultiplyPoint3x4(vertex);if(!found){result=new UnityEngine.Bounds(p,UnityEngine.Vector3.zero);found=true;}else result.Encapsulate(p);}
    }
    if(!found)throw new System.InvalidOperationException("Нет видимой mesh geometry: "+prefab.name);
    return result;
};
