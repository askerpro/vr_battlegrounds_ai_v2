System.Func<string> loadedSceneFingerprint = () => {
    var snapshot = new System.Text.StringBuilder();
    for (int sceneIndex=0; sceneIndex<UnityEngine.SceneManagement.SceneManager.sceneCount; sceneIndex++) {
        var loadedScene=UnityEngine.SceneManagement.SceneManager.GetSceneAt(sceneIndex);
        if (!loadedScene.isLoaded || UnityEditor.SceneManagement.EditorSceneManager.IsPreviewScene(loadedScene)) continue;
        snapshot.AppendLine(loadedScene.path+"|dirty="+loadedScene.isDirty);
        foreach (var sceneRoot in loadedScene.GetRootGameObjects()) {
            snapshot.AppendLine("ROOT|"+sceneRoot.GetInstanceID());
            foreach (var node in sceneRoot.GetComponentsInChildren<UnityEngine.Transform>(true)) {
                snapshot.AppendLine("GO|"+node.gameObject.GetInstanceID()+"|"+UnityEditor.EditorJsonUtility.ToJson(node.gameObject));
                foreach (var component in node.GetComponents<UnityEngine.Component>())
                    if (component!=null) snapshot.AppendLine("COMP|"+component.GetInstanceID()+"|"+UnityEditor.EditorJsonUtility.ToJson(component));
            }
        }
    }
    using(var sha=System.Security.Cryptography.SHA256.Create())
        return System.BitConverter.ToString(sha.ComputeHash(System.Text.Encoding.UTF8.GetBytes(snapshot.ToString())));
};

        string assetPath="Assets/Prefabs/Weapons/"+id+"/"+id+".prefab";
        var prefabAsset=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.GameObject>(assetPath);
        snapshot.AppendLine(assetPath+"|"+UnityEditor.AssetDatabase.GetAssetDependencyHash(assetPath));
        foreach(var renderer in prefabAsset.GetComponentsInChildren<UnityEngine.Renderer>(true))
            foreach(var material in renderer.sharedMaterials)
                if(material!=null && capturedMaterials.Add(material.GetInstanceID()))
                    snapshot.AppendLine(material.GetInstanceID()+"|dirty="+UnityEditor.EditorUtility.IsDirty(material)+"|"+UnityEditor.EditorJsonUtility.ToJson(material));
    }
    using(var sha=System.Security.Cryptography.SHA256.Create())
        return System.BitConverter.ToString(sha.ComputeHash(System.Text.Encoding.UTF8.GetBytes(snapshot.ToString())));
};
