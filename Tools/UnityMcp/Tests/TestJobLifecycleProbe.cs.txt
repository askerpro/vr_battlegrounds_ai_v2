namespace AgentInfra.TestJobLifecycleProbe
{
    // Временный инфраструктурный стенд: намеренный Failed проверяет доставку результата.
    // Устанавливается только в собственной аренде; finish сохраняет .meta и убирает стенд с worker.
    [PrebuildSetup(typeof(LongPrebuild))]
    public sealed class LifecycleTests
    {
        [Test, Order(1)]
        public void ActivePrebuildKeepsJobAttached()
        {
            var receipt = JsonUtility.FromJson<PrebuildReceipt>(File.ReadAllText(LongPrebuild.ReportPath));
            Assert.That(receipt.job_id, Is.EqualTo(ProbeState.CurrentJobId()), "Receipt принадлежит другому запуску.");
            Assert.That(receipt.passed, Is.True, JsonUtility.ToJson(receipt));
        }

        [Test, Order(0)]
        public void IntentionalFailureReturnsFullResult()
        {
            Assert.Fail("AGENT_INFRA_EXPECTED_FAILURE_RESULT_PROBE");
        }

        [UnityTest, Order(2)]
        public IEnumerator MidrunReloadRetainsJobAndEarlierResults()
        {
            var id = ProbeState.CurrentJobId();
            var prefix = ProbeState.ReloadKey(id);
            // При replay IEnumerator после reload повторный запрос компиляции запрещён.
            if (!SessionState.GetBool(prefix + ".requested", false))
            {
                SessionState.SetString(prefix + ".job", id);
                SessionState.SetString(prefix + ".domain_before", ProbeState.DomainIdentity);
                SessionState.SetBool(prefix + ".requested", true);
                CompilationPipeline.RequestScriptCompilation();
            }
            yield return new RecompileScripts();
            // После reload не полагаемся на locals старого iterator: identity хранится в SessionState.
            var receiptId = SessionState.GetString(ProbeState.CurrentProbeJobKey, string.Empty);
            var restoredPrefix = ProbeState.ReloadKey(receiptId);
            var savedId = SessionState.GetString(restoredPrefix + ".job", string.Empty);
            var before = SessionState.GetString(restoredPrefix + ".domain_before", string.Empty);
            var after = ProbeState.DomainIdentity;
            var current = ProbeState.CurrentJobId();
            var receipt = new ReloadReceipt
            {
                job_id = receiptId, session_job_id = savedId, current_job_id_after = current,
                domain_before = before, domain_after = after,
                passed = savedId == receiptId && current == receiptId && !string.IsNullOrEmpty(before) && before != after
            };
            var path = LongPrebuild.PathForJob(receiptId, "reload-receipt.json");
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            File.WriteAllText(path, JsonUtility.ToJson(receipt, true));
            Assert.That(receipt.passed, Is.True, JsonUtility.ToJson(receipt));
        }
    }

    [Serializable]
    public sealed class PrebuildReceipt
    {
        public bool passed;
        public string job_id;
        public string current_job_id_after;
        public bool native_active;
        public long elapsed_ms;
        public string status;
        public bool total_is_null;
        public string error;
    }

    [Serializable]
    public sealed class ReloadReceipt
    {
        public bool passed;
        public string job_id, session_job_id, current_job_id_after, domain_before, domain_after;
    }

    internal static class ProbeState
    {
        public static readonly string DomainIdentity = Guid.NewGuid().ToString("N");
        public const string CurrentProbeJobKey = "AgentInfra.TestJobLifecycleProbe.CurrentJob";
        public static string ReloadKey(string id) => "AgentInfra.TestJobLifecycleProbe." + id + ".Reload";
        public static string CurrentJobId()
        {
            const BindingFlags flags = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;
            var manager = AppDomain.CurrentDomain.GetAssemblies()
                .Select(assembly => assembly.GetType("MCPForUnity.Editor.Services.TestJobManager", false))
                .First(type => type != null);
            return (string)manager.GetProperty("CurrentJobId", flags).GetValue(null);
        }
    }

    public sealed class LongPrebuild : IPrebuildSetup
    {
        public static string ReportPath => PathForJob(
            SessionState.GetString(ProbeState.CurrentProbeJobKey, string.Empty), "prebuild-receipt.json");
        public static string PathForJob(string id, string filename)
        {
            if (!Guid.TryParse(id, out var parsed)) throw new InvalidOperationException("Receipt требует текущий job_id.");
            return Path.GetFullPath("tasks/agent-coordination-protocol/reports/mcp-test-job-lifecycle-20261015/"
                + parsed.ToString("N") + "/" + filename);
        }

        public void Setup()
        {
            const BindingFlags flags = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;
            var manager = AppDomain.CurrentDomain.GetAssemblies()
                .Select(assembly => assembly.GetType("MCPForUnity.Editor.Services.TestJobManager", false))
                .First(type => type != null);
            var current = manager.GetProperty("CurrentJobId", flags);
            var id = (string)current.GetValue(null);
            if (string.IsNullOrEmpty(id)) throw new InvalidOperationException("Стенд запускается только через MCP job.");
            SessionState.SetString(ProbeState.CurrentProbeJobKey, id);

            // Реальная native prebuild-фаза превышает стандартные 15 с до RunStarted.
            // Этот единственный ограниченный sleep не запускает и не меняет игровую механику.
            Thread.Sleep(17000);
            var native = typeof(TestRunnerApi).GetMethod("IsRunActive", flags, null, Type.EmptyTypes, null);
            var active = native != null && (bool)native.Invoke(null, null);
            var job = manager.GetMethod("GetJob", flags).Invoke(null, new object[] { id });
            var jobType = job.GetType();
            var receipt = new PrebuildReceipt
            {
                job_id = id,
                current_job_id_after = (string)current.GetValue(null),
                native_active = active,
                elapsed_ms = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()
                    - (long)jobType.GetProperty("StartedUnixMs").GetValue(job),
                status = jobType.GetProperty("Status").GetValue(job).ToString(),
                total_is_null = jobType.GetProperty("TotalTests").GetValue(job) == null,
                error = (string)jobType.GetProperty("Error").GetValue(job)
            };
            receipt.passed = receipt.native_active && receipt.elapsed_ms > 15000
                && receipt.status == "Running" && receipt.total_is_null
                && receipt.current_job_id_after == id && string.IsNullOrEmpty(receipt.error);
            Directory.CreateDirectory(Path.GetDirectoryName(ReportPath));
            File.WriteAllText(ReportPath, JsonUtility.ToJson(receipt, true));
        }
    }
}
