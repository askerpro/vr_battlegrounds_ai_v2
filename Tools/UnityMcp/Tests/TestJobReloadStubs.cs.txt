namespace UnityEditor.TestTools.TestRunner.Api
{
    public enum TestMode { EditMode, PlayMode }
    public class Filter { public TestMode testMode; public string[] testNames,groupNames,categoryNames,assemblyNames; }
    public class ExecutionSettings { public Filter filter; public ExecutionSettings(Filter f) { filter=f; } }
    public interface ITestAdaptor { bool HasChildren { get; } bool IsSuite { get; } IEnumerable<ITestAdaptor> Children { get; } string Name { get; } string FullName { get; } }
    public interface ITestResultAdaptor
    {
        bool HasChildren { get; } IEnumerable<ITestResultAdaptor> Children { get; } ITestAdaptor Test { get; } int PassCount { get; } int FailCount { get; } int SkipCount { get; }
        string Name { get; } string FullName { get; } string ResultState { get; } double Duration { get; } string Message { get; } string StackTrace { get; } string Output { get; }
    }
    public interface ICallbacks { void RunStarted(ITestAdaptor tests); void RunFinished(ITestResultAdaptor result); void TestStarted(ITestAdaptor test); void TestFinished(ITestResultAdaptor result); }
    public class TestRunnerApi
    {
        public static bool Active;
        public static int ExecuteCalls;
        public static readonly List<ICallbacks> Callbacks=new();
        internal static bool IsRunActive() => Active;
        public void RegisterCallbacks(ICallbacks receiver) => Callbacks.Add(receiver);
        public void UnregisterCallbacks(ICallbacks receiver) => Callbacks.Remove(receiver);
        public string Execute(ExecutionSettings settings) { ExecuteCalls++; Active=true; return "fixture-native-guid"; }
        public void RetrieveTestList(TestMode mode,Action<ITestAdaptor> callback) => callback(new Node("fixture",false));
        // События — внешняя граница framework. Они не синтезируются production readiness.
        public static void Complete(bool failed)
        {
            var one=new Node("Fixture.One",false); var two=new Node("Fixture.Two",false);
            var root=new Node("Fixture",true,new[] {one,two});
            foreach(var receiver in Callbacks.ToArray()) receiver.RunStarted(root);
            foreach(var item in new[] {new Result(one,false),new Result(two,failed)})
                foreach(var receiver in Callbacks.ToArray()) { receiver.TestStarted(item.Test); receiver.TestFinished(item); }
            Active=false;
            var summary=new Result(root,failed) { PassCount=failed?1:2, FailCount=failed?1:0, Children=new[] {new Result(one,false),new Result(two,failed)} };
            foreach(var receiver in Callbacks.ToArray()) receiver.RunFinished(summary);
        }
    }
    public static class MidrunBoundary
    {
        public static Node First() => new Node("Fixture.Parameterized(1)",false);
        public static Node Last() => new Node("Fixture.Parameterized(2)",false);
        public static Node Suite() => new Node("Fixture.Parameterized",true,new[] {First(),Last()});
        public static void BeforeReload()
        {
            foreach(var receiver in TestRunnerApi.Callbacks.ToArray())
            {
                receiver.RunStarted(Suite());receiver.TestStarted(First());receiver.TestFinished(new Result(First(),true));
            }
        }
        public static void FinishAfterReload()
        {
            var last=new Result(Last(),false) {ResultState="Skipped",Message="fixture skip"};
            foreach(var receiver in TestRunnerApi.Callbacks.ToArray()) {receiver.TestStarted(last.Test);receiver.TestFinished(last);}
            var old=new Result(First(),true);
            var nested=new Result(Suite(),true) {Children=new[] {old,last},FailCount=1,SkipCount=1};
            var rootNode=new Node("Fixture.Root",true,new[] {Suite()});
            var terminal=new Result(rootNode,true) {Children=new[] {nested},FailCount=1,SkipCount=1};
            TestRunnerApi.Active=false;
            foreach(var receiver in TestRunnerApi.Callbacks.ToArray()) receiver.RunFinished(terminal);
        }
        public static void EmptySuite()
        {
            var node=new Node("Fixture.EmptySuite",false,isSuite:true);
            var terminal=new Result(node,false);
            TestRunnerApi.Active=false;
            foreach(var receiver in TestRunnerApi.Callbacks.ToArray()) receiver.RunFinished(terminal);
        }
    }
    public class Node : ITestAdaptor
    {
        public Node(string name,bool children,IEnumerable<ITestAdaptor> items=null,bool? isSuite=null) {Name=name;HasChildren=children;Children=items;IsSuite=isSuite??children;}
        public bool HasChildren {get;} public bool IsSuite {get;} public IEnumerable<ITestAdaptor> Children {get;} public string Name {get;} public string FullName => Name;
    }
    public class Result : ITestResultAdaptor
    {
        public Result(ITestAdaptor test,bool failed) { Test=test; ResultState=failed?"Failed":"Passed"; Message=failed?"fixture assertion":""; }
        public bool HasChildren=>Test.HasChildren; public IEnumerable<ITestResultAdaptor> Children {get;set;} public ITestAdaptor Test {get;} public int PassCount {get;set;} public int FailCount {get;set;} public int SkipCount {get;set;}
        public string Name=>Test.Name; public string FullName=>Test.FullName; public string ResultState {get;set;} public double Duration=>0.25; public string Message {get;set;} public string StackTrace=>"fixture stack"; public string Output=>"fixture output";
    }
}
