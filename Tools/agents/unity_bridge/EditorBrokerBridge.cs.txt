// Локальный мост без зависимости от игровой сборки. Состояние хранится вне checkout.

{
    [Serializable] internal sealed class BridgeConfig
    {
        public int version = 0;
        public string project_root = null;
        public string state_dir = null;
    }

    [Serializable] internal sealed class BridgeScene
    {
        public string path;
        public bool is_loaded;
        public bool is_active;
    }

    [Serializable] internal sealed class BridgeSetup
    {
        public BridgeScene[] scenes = null;
        public string prefab_path = null;
        public string prefab_asset_path = null; // Совместимость с ранним Python-адаптером.
    }

    [Serializable] internal sealed class BridgeRequest
    {
        public int version = 0;
        public string request_id = null;
        public string project_root = null;
        public string operation = null;
        public long created_unix_ms = 0;
        public string[] output_roots = null;
        public BridgeSetup setup = null;
    }

    [Serializable] internal sealed class BridgeResult
    {
        public string project_root;
        public bool ready;
        public bool is_playing;
        public bool is_compiling;
        public bool is_updating;
        public bool is_test_running;
        public string[] dirty_scenes;
        public string[] dirty_assets;
        public string[] ignored_dirty_assets;
        public string prefab_stage;
        public int process_id;
        public string session_id;
        public bool auto_refresh_suppressed;
        public bool compilation_failed;
        public string[] compile_errors;
        public BridgeScene[] scenes;
        public string prefab_path;
        public string[] saved_paths;
    }

    [Serializable] internal sealed class BridgeJournal
    {
        public BridgeRequest request;
        public string session_id;
        public string stage;
        public BridgeResult result;
        public string error;
    }

    [Serializable] internal sealed class BridgeResponse
    {
        public string request_id;
        public string project_root;
        public bool ok;
        public string error;
        public BridgeResult result;
    }

    [Serializable] internal sealed class BridgeHeartbeat
    {
        public string project_root;
        public int process_id;
        public string session_id;
        public long updated_unix_ms;
        public string error;
        public long journal_reads;
        public long request_reads;
        public long archive_reads;
    }

    [InitializeOnLoad]
    public static class EditorBrokerBridge
    {
        private const string LocalPath = "Assets/Editor/VR_Battlegrounds/Debug/EditorBrokerLocal";
        private const string SessionKey = "EditorBrokerLocal.Session";
        private const string RefreshKey = "EditorBrokerLocal.AutoRefreshSuppressed";
        private static readonly string Root = Path.GetFullPath(Path.Combine(Application.dataPath, ".."));
        private static readonly string Session = GetSession();
        private static BridgeConfig _config;
        private static string _mailbox;
        private static BridgeJournal _current;
        private static double _nextPoll;
        private static double _nextHeartbeat;
        private static int _idleTicks;
        private static bool _terminalPersisted;
        private static long _journalReads;
        private static long _requestReads;
        private static long _archiveReads;
        private static readonly List<string> CompileErrors = new List<string>();

        static EditorBrokerBridge()
        {
            EditorApplication.update += Tick;
            CompilationPipeline.compilationStarted += context => CompileErrors.Clear();
            CompilationPipeline.assemblyCompilationFinished += (assembly, messages) => {
                CompileErrors.AddRange(messages.Where(message => message.type == CompilerMessageType.Error)
                    .Select(message => message.file + ":" + message.line + ": " + message.message));
            };
        }

        private static string GetSession()
        {
            string value = SessionState.GetString(SessionKey, "");
            if (value.Length == 0)
            {
                value = Guid.NewGuid().ToString("N");
                SessionState.SetString(SessionKey, value);
            }
            return value;
        }

        private static bool SamePath(string left, string right)
        {
            return string.Equals(Path.GetFullPath(left).TrimEnd('/', '\\'),
                Path.GetFullPath(right).TrimEnd('/', '\\'), StringComparison.OrdinalIgnoreCase);
        }

        private static bool Under(string path, string parent)
        {
            return Path.GetFullPath(path).StartsWith(Path.GetFullPath(parent).TrimEnd('/', '\\') +
                Path.DirectorySeparatorChar, StringComparison.OrdinalIgnoreCase);
        }

        private static void AtomicJson(string path, object value)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            string temporary = path + "." + Guid.NewGuid().ToString("N") + ".tmp";
            byte[] bytes = System.Text.Encoding.UTF8.GetBytes(JsonUtility.ToJson(value, true));
            try
            {
                using (var stream = new FileStream(temporary, FileMode.CreateNew, FileAccess.Write, FileShare.None))
                {
                    stream.Write(bytes, 0, bytes.Length);
                    stream.Flush(true);
                }
                if (File.Exists(path)) File.Replace(temporary, path, null);
                else File.Move(temporary, path);
            }
            finally
            {
                // Удаляется только временный файл этой попытки, никогда mailbox/journal назначения.
                try { if (File.Exists(temporary)) File.Delete(temporary); } catch { }
            }
        }

        private static void Configure()
        {
            string path = Path.Combine(Root, LocalPath, "bridge-config.json");
            if (!File.Exists(path)) return;
            var config = JsonUtility.FromJson<BridgeConfig>(File.ReadAllText(path));
            if (config == null || config.version != 1 || !SamePath(config.project_root, Root))
                throw new InvalidOperationException("bridge-config project_root/version не совпадает");
            string state = Path.GetFullPath(config.state_dir);
            if (SamePath(state, Root) || Under(state, Root))
                throw new InvalidOperationException("state_dir находится внутри переключаемого checkout");
            _config = config;
            _mailbox = Path.Combine(state, "unity");
            foreach (string name in new[] { "requests", "responses", "journals" })
                Directory.CreateDirectory(Path.Combine(_mailbox, name));
            Directory.CreateDirectory(Path.Combine(_mailbox, "archive", "requests"));
        }

        private static void Heartbeat(string error = "")
        {
            if (_mailbox == null) return;
            AtomicJson(Path.Combine(_mailbox, "heartbeat.json"), new BridgeHeartbeat {
                project_root = Root, process_id = System.Diagnostics.Process.GetCurrentProcess().Id,
                session_id = Session, updated_unix_ms = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds(), error = error,
                journal_reads = _journalReads, request_reads = _requestReads, archive_reads = _archiveReads });
        }

        private static bool TestsRunning()
        {
            // В установленном Test Framework 1.4 метод internal static IsRunActive().
            // Неизвестная версия API закрывает изменения, пока её явно не проверят.
            foreach (System.Reflection.Assembly assembly in AppDomain.CurrentDomain.GetAssemblies())
            {
                Type type = assembly.GetType("UnityEditor.TestTools.TestRunner.Api.TestRunnerApi", false);
                if (type == null) continue;
                MethodInfo method = type.GetMethod("IsRunActive", BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public);
                if (method == null || method.ReturnType != typeof(bool)) return true;
                try { return (bool)method.Invoke(null, null); }
                catch { return true; }
            }
            return false;
        }

        private static BridgeResult InspectState()
        {
            var result = new BridgeResult {
                project_root = Root, is_playing = EditorApplication.isPlayingOrWillChangePlaymode,
                is_compiling = EditorApplication.isCompiling, is_updating = EditorApplication.isUpdating,
                is_test_running = TestsRunning(), dirty_scenes = new string[0], dirty_assets = new string[0],
                ignored_dirty_assets = new string[0],
                prefab_stage = "", prefab_path = "", scenes = new BridgeScene[0],
                process_id = System.Diagnostics.Process.GetCurrentProcess().Id, session_id = Session,
                auto_refresh_suppressed = SessionState.GetBool(RefreshKey, false),
                compilation_failed = EditorUtility.scriptCompilationFailed,
                compile_errors = !EditorUtility.scriptCompilationFailed ? new string[0] :
                    CompileErrors.Count != 0 ? CompileErrors.ToArray() : new[] { "Unity сообщает scriptCompilationFailed" } };
            // Play/переход Play наблюдаем через флаги: GetSceneManagerSetup разрешён только в Edit Mode.
            // ready означает завершённую операцию Editor; dirty/compile_errors проверяет владелец перехода.
            result.ready = !Busy(result);
            if (result.is_playing) return result;
            var dirtyScenes = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            for (int index = 0; index < SceneManager.sceneCount; index++)
            {
                Scene scene = SceneManager.GetSceneAt(index);
                if (scene.isDirty || string.IsNullOrEmpty(scene.path) && scene.rootCount > 0)
                    dirtyScenes.Add(string.IsNullOrEmpty(scene.path) ? "<unsaved-scene>" : scene.path);
            }
            PrefabStage prefab = PrefabStageUtility.GetCurrentPrefabStage();
            if (prefab != null && prefab.scene.isDirty) dirtyScenes.Add(prefab.assetPath);
            var dirtyAssets = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            var ignoredAssets = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (UnityEngine.Object item in Resources.FindObjectsOfTypeAll<UnityEngine.Object>())
            {
                if (item == null || !EditorUtility.IsPersistent(item) || !EditorUtility.IsDirty(item)) continue;
                string path = AssetDatabase.GetAssetPath(item);
                if (string.IsNullOrEmpty(path) || !path.StartsWith("Assets/", StringComparison.OrdinalIgnoreCase)) continue;
                // Несохранённое — только то, что пишет SaveAssets: нативные ассеты и настройки импорта.
                // Объекты импорта (атлас динамического шрифта, шейдер .shader, меш FBX) Unity помечает
                // заново при работе, а сохранить их нельзя: они лишь перечисляются для диагностики.
                if (item is AssetImporter || AssetDatabase.IsNativeAsset(item)) dirtyAssets.Add(path);
                else ignoredAssets.Add(path);
            }
            result.dirty_scenes = dirtyScenes.OrderBy(path => path).ToArray();
            result.dirty_assets = dirtyAssets.OrderBy(path => path).ToArray();
            result.ignored_dirty_assets = ignoredAssets.Except(dirtyAssets, StringComparer.OrdinalIgnoreCase)
                .OrderBy(path => path).ToArray();
            result.prefab_stage = prefab == null ? "" : prefab.assetPath;
            BridgeResult setup = SceneSetup();
            result.scenes = setup.scenes;
            result.prefab_path = setup.prefab_path;
            return result;
        }

        private static bool Busy(BridgeResult result)
        {
            return result.is_playing || result.is_compiling || result.is_updating || result.is_test_running;
        }

        private static void RequireClean(BridgeResult result)
        {
            if (Busy(result)) throw new InvalidOperationException("Unity занят: Play/compile/import/tests");
            if (result.dirty_scenes.Length != 0 || result.dirty_assets.Length != 0)
                throw new InvalidOperationException("Несохранённые сцены/ассеты; изменения запрещены");
        }

        private static string JournalPath(string id) { return Path.Combine(_mailbox, "journals", id + ".json"); }
        private static string ResponsePath(string id) { return Path.Combine(_mailbox, "responses", id + ".json"); }
        private static string ArchivePath(string id) { return Path.Combine(_mailbox, "archive", id + ".json"); }
        private static void Persist(BridgeJournal journal) { AtomicJson(JournalPath(journal.request.request_id), journal); }

        private static BridgeJournal ReadArchived(string id)
        {
            _archiveReads++;
            var journal = JsonUtility.FromJson<BridgeJournal>(File.ReadAllText(ArchivePath(id)));
            if (journal == null || journal.stage != "complete" || journal.request == null ||
                journal.request.version != 1 || journal.request.request_id != id || !SamePath(journal.request.project_root, Root))
                throw new InvalidOperationException("Архив запроса повреждён или принадлежит другому root/version");
            return journal;
        }

        private static void ArchiveRequest(string id)
        {
            string source = Path.Combine(_mailbox, "requests", id + ".json");
            if (!File.Exists(source)) return;
            string destination = Path.Combine(_mailbox, "archive", "requests", id + ".json");
            // Повторный вход сохраняется отдельно, исходная immutable запись не заменяется.
            if (File.Exists(destination)) destination = Path.Combine(_mailbox, "archive", "requests", id + "." + Guid.NewGuid().ToString("N") + ".json");
            File.Move(source, destination);
        }

        private static void ArchiveCompleted(BridgeJournal journal)
        {
            string id = journal.request.request_id;
            string source = JournalPath(id), destination = ArchivePath(id);
            if (journal.stage != "complete") throw new InvalidOperationException("Архивируется только terminal запрос");
            // Вызывается только ПОСЛЕ durable terminal и успешной доставки response.
            if (!File.Exists(destination)) File.Move(source, destination);
            else if (File.Exists(source))
            {
                if (File.ReadAllText(source) != File.ReadAllText(destination))
                    throw new InvalidOperationException("Конфликт immutable архива запроса");
                File.Delete(source);
            }
            ArchiveRequest(id);
        }

        private static bool ValidId(string id)
        {
            return id != null && id.Length == 32 && id.All(c => "0123456789abcdef".IndexOf(c) >= 0);
        }

        private static void Reply(BridgeJournal journal)
        {
            AtomicJson(ResponsePath(journal.request.request_id), new BridgeResponse {
                request_id = journal.request.request_id, project_root = Root,
                ok = string.IsNullOrEmpty(journal.error), error = journal.error ?? "", result = journal.result });
        }

        private static void Complete(BridgeJournal journal, string error = "")
        {
            journal.error = error;
            journal.stage = "complete";
            _terminalPersisted = false;
            Persist(journal); // Терминальная запись не позволяет replay после удаления ответа.
            _terminalPersisted = true;
            Reply(journal);
            ArchiveCompleted(journal);
            _current = null;
            _idleTicks = 0;
        }

        private static BridgeJournal NextRequest()
        {
            // Журнал переживает reload независимо от наличия/удаления request-файла.
            foreach (string path in Directory.GetFiles(Path.Combine(_mailbox, "journals"), "*.json"))
            {
                _journalReads++;
                var journal = JsonUtility.FromJson<BridgeJournal>(File.ReadAllText(path));
                if (journal.stage != "complete" && File.Exists(ResponsePath(journal.request.request_id)))
                {
                    // Recovery остановленного Editor может завершить запрос внешним ответом.
                    var response = JsonUtility.FromJson<BridgeResponse>(File.ReadAllText(ResponsePath(journal.request.request_id)));
                    journal.stage = "complete";
                    journal.error = response.error;
                    journal.result = response.result;
                    Persist(journal);
                    ArchiveCompleted(journal);
                    continue;
                }
                if (journal.stage != "complete") return journal;
                if (!File.Exists(ResponsePath(journal.request.request_id))) Reply(journal);
                ArchiveCompleted(journal);
            }
            BridgeRequest next = null;
            foreach (string path in Directory.GetFiles(Path.Combine(_mailbox, "requests"), "*.json"))
            {
                _requestReads++;
                var request = JsonUtility.FromJson<BridgeRequest>(File.ReadAllText(path));
                if (request == null || !ValidId(request.request_id) || Path.GetFileNameWithoutExtension(path) != request.request_id)
                    continue;
                if (File.Exists(ArchivePath(request.request_id)))
                {
                    Reply(ReadArchived(request.request_id));
                    ArchiveRequest(request.request_id);
                    continue;
                }
                if (File.Exists(JournalPath(request.request_id))) continue;
                if (File.Exists(ResponsePath(request.request_id)))
                {
                    var response = JsonUtility.FromJson<BridgeResponse>(File.ReadAllText(ResponsePath(request.request_id)));
                    var cancelled = new BridgeJournal { request = request, session_id = Session, stage = "complete",
                        result = response.result, error = response.error };
                    Persist(cancelled);
                    ArchiveCompleted(cancelled);
                    continue;
                }
                if (next == null || request.created_unix_ms < next.created_unix_ms) next = request;
            }
            if (next == null) return null;
            var value = new BridgeJournal { request = next, session_id = Session, stage = "queued" };
            Persist(value);
            return value;
        }

        private static void Tick()
        {
            if (EditorApplication.timeSinceStartup < _nextPoll) return;
            _nextPoll = EditorApplication.timeSinceStartup + 0.1;
            try
            {
                if (_config == null) Configure();
                if (_config == null) return;
                if (EditorApplication.timeSinceStartup >= _nextHeartbeat)
                {
                    Heartbeat();
                    _nextHeartbeat = EditorApplication.timeSinceStartup + 1;
                }
                if (File.Exists(Path.Combine(_mailbox, "recovery-block.json"))) return;
                if (_current == null) _current = NextRequest();
                if (_current == null) return;
                if (_current.stage == "complete")
                {
                    // Сначала подтверждаем durable terminal: File.Replace мог отказать из-за чужого reader.
                    // Повторяется только запись результата, а не уже выполненное изменение Unity.
                    if (!_terminalPersisted)
                    {
                        Persist(_current);
                        _terminalPersisted = true;
                    }
                    Reply(_current);
                    ArchiveCompleted(_current);
                    _current = null;
                    return;
                }
                BridgeRequest request = _current.request;
                if (request.version != 1 || !SamePath(request.project_root, Root))
                    throw new InvalidOperationException("request project_root/version не совпадает");
                if (_current.session_id != Session)
                    throw new InvalidOperationException("Запрос принадлежит завершённому Editor process; требуется recovery");
                if (_current.stage == "dispatched")
                    throw new InvalidOperationException("Прервано изменение Unity; неизвестен итог. Повтор запрещён, требуется recovery");
                if (_current.stage == "waiting")
                {
                    BridgeResult state = InspectState();
                    if (Busy(state)) { _idleTicks = 0; return; }
                    // Два последовательных update после reload/Refresh защищают от преждевременного ответа.
                    if (++_idleTicks < 3) return;
                    RequireClean(state);
                    if (request.operation == "park" && (!state.auto_refresh_suppressed || SceneManager.sceneCount != 1 ||
                        SceneManager.GetSceneAt(0).path != "" || PrefabStageUtility.GetCurrentPrefabStage() != null))
                        throw new InvalidOperationException("Состояние park не подтверждено");
                    if (request.operation != "park" && state.auto_refresh_suppressed)
                        throw new InvalidOperationException("AutoRefresh остался подавлен; требуется recovery");
                    BridgeResult output = _current.result;
                    if (request.operation == "park" && output != null)
                    {
                        state.scenes = output.scenes;
                        state.prefab_path = output.prefab_path;
                    }
                    state.saved_paths = output == null ? null : output.saved_paths;
                    _current.result = state;
                    Complete(_current);
                    return;
                }
                Dispatch(_current);
            }
            catch (Exception exception)
            {
                try
                {
                    if (_current != null && _current.stage != "complete") Complete(_current, exception.Message);
                    else Heartbeat(exception.Message);
                }
                catch { /* IO-сбой оставляет durable journal; следующая загрузка не повторяет изменение. */ }
            }
        }

        private static BridgeResult SceneSetup()
        {
            PrefabStage prefab = PrefabStageUtility.GetCurrentPrefabStage();
            return new BridgeResult {
                scenes = EditorSceneManager.GetSceneManagerSetup().Select(scene => new BridgeScene {
                    path = scene.path, is_loaded = scene.isLoaded, is_active = scene.isActive }).ToArray(),
                prefab_path = prefab == null ? "" : prefab.assetPath };
        }

        private static void Dispatch(BridgeJournal journal)
        {
            BridgeRequest request = journal.request;
            BridgeResult state = InspectState();
            if (request.operation == "inspect") { journal.result = state; Complete(journal); return; }
            if (Busy(state)) throw new InvalidOperationException("Unity занят: Play/compile/import/tests");
            if (request.operation != "save_outputs") RequireClean(state);
            if (request.operation == "park" && state.auto_refresh_suppressed)
                throw new InvalidOperationException("Editor уже припаркован");
            if (request.operation != "park" && request.operation != "refresh" && state.auto_refresh_suppressed)
                throw new InvalidOperationException("Сначала требуется refresh припаркованного Editor");
            // Все проверки выполняются до durable intent и изменения Unity.
            if (request.operation == "restore") ValidateSetup(request.setup);
            if (request.operation == "save_outputs") ValidateOutputs(request.output_roots, state);
            if (!new[] { "park", "refresh", "save_outputs", "restore" }.Contains(request.operation))
                throw new InvalidOperationException("Недопустимая операция");
            if (request.operation == "park")
            {
                journal.result = SceneSetup();
                ValidateSetup(new BridgeSetup { scenes = journal.result.scenes, prefab_path = journal.result.prefab_path });
            }
            journal.stage = "dispatched";
            Persist(journal);
            if (request.operation == "park")
            {
                if (PrefabStageUtility.GetCurrentPrefabStage() != null) StageUtility.GoToMainStage();
                EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
                AssetDatabase.DisallowAutoRefresh();
                SessionState.SetBool(RefreshKey, true);
            }
            else if (request.operation == "refresh")
            {
                // Запись waiting ДО вызова, который может перезагрузить домен.
                journal.stage = "waiting";
                Persist(journal);
                if (SessionState.GetBool(RefreshKey, false))
                {
                    SessionState.SetBool(RefreshKey, false);
                    AssetDatabase.AllowAutoRefresh();
                }
                AssetDatabase.Refresh(ImportAssetOptions.ForceUpdate);
                return;
            }
            else if (request.operation == "save_outputs") journal.result = SaveOutputs(request.output_roots, state);
            else RestoreSetup(request.setup);
            journal.stage = "waiting";
            Persist(journal);
        }

        private static string ValidateAssetPath(string path)
        {
            if (string.IsNullOrEmpty(path) || path.IndexOf('\\') >= 0 || path.IndexOf(':') >= 0 ||
                Path.IsPathRooted(path) || path.Split('/').Any(p => p == "" || p == "." || p == ".."))
                throw new InvalidOperationException("Недопустимый путь ассета");
            if (path != "Assets" && !path.StartsWith("Assets/", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("Сохранение разрешено только внутри Assets");
            if (string.Equals(path, LocalPath, StringComparison.OrdinalIgnoreCase) ||
                path.StartsWith(LocalPath + "/", StringComparison.OrdinalIgnoreCase) ||
                string.Equals(path, LocalPath + ".meta", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("Локальный мост не является выходным ассетом");
            string fullPath = Path.Combine(Root, path);
            if (!Under(fullPath, Root)) throw new InvalidOperationException("Путь выходит за project root");
            // Reparse points запрещены: симлинк внутри Assets не должен писать вне worktree.
            for (string item = fullPath; !SamePath(item, Root); item = Path.GetDirectoryName(item))
                if ((Directory.Exists(item) || File.Exists(item)) &&
                    (File.GetAttributes(item) & FileAttributes.ReparsePoint) != 0)
                    throw new InvalidOperationException("Выходной путь содержит reparse point");
            return path;
        }

        private static bool Included(string path, string[] roots)
        {
            return roots.Any(root => string.Equals(path, root, StringComparison.OrdinalIgnoreCase) ||
                path.StartsWith(root + "/", StringComparison.OrdinalIgnoreCase));
        }

        private static void ValidateOutputs(string[] roots, BridgeResult state)
        {
            if (roots == null) throw new InvalidOperationException("Нет output_roots");
            // Roots исходников вне Assets допустимы для Git; Unity сохраняет только Assets.
            foreach (string path in roots)
            {
                if (string.IsNullOrEmpty(path) || Path.IsPathRooted(path) || path.IndexOf(':') >= 0 ||
                    path.IndexOf('\\') >= 0 || path.Split('/').Any(p => p == "" || p == "." || p == ".."))
                    throw new InvalidOperationException("Недопустимый output root");
                if (path == "Assets" || path.StartsWith("Assets/", StringComparison.OrdinalIgnoreCase)) ValidateAssetPath(path);
                else if (new[] { ".git", "Library", "Temp", "Logs", "obj" }.Any(p =>
                    string.Equals(path, p, StringComparison.OrdinalIgnoreCase) || path.StartsWith(p + "/", StringComparison.OrdinalIgnoreCase)))
                    throw new InvalidOperationException("Запрещённый output root");
            }
            if (!string.IsNullOrEmpty(state.prefab_stage))
                throw new InvalidOperationException("Закройте prefab stage до save_outputs");
            foreach (string path in state.dirty_scenes.Concat(state.dirty_assets))
            {
                ValidateAssetPath(path);
                if (!Included(path, roots)) throw new InvalidOperationException("Чужой dirty-объект: " + path);
            }
        }

        private static BridgeResult SaveOutputs(string[] roots, BridgeResult state)
        {
            var saved = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (string path in state.dirty_assets)
            {
                string guid = AssetDatabase.AssetPathToGUID(path);
                if (string.IsNullOrEmpty(guid)) throw new InvalidOperationException("Dirty-ассет отсутствует: " + path);
                AssetDatabase.SaveAssetIfDirty(new GUID(guid));
                saved.Add(path);
            }
            for (int index = 0; index < SceneManager.sceneCount; index++)
            {
                Scene scene = SceneManager.GetSceneAt(index);
                if (!scene.isDirty) continue;
                if (!EditorSceneManager.SaveScene(scene)) throw new InvalidOperationException("Не удалось сохранить сцену: " + scene.path);
                saved.Add(scene.path);
            }
            return new BridgeResult { saved_paths = saved.OrderBy(path => path).ToArray() };
        }

        private static void ValidateSetup(BridgeSetup setup)
        {
            if (setup == null || setup.scenes == null) throw new InvalidOperationException("Нет setup.scenes");
            int loaded = 0, active = 0;
            var paths = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (BridgeScene scene in setup.scenes)
            {
                if (scene == null) throw new InvalidOperationException("Пустая запись сцены");
                if (string.IsNullOrEmpty(scene.path))
                {
                    if (setup.scenes.Length != 1 || !scene.is_loaded || !scene.is_active)
                        throw new InvalidOperationException("Пустая сцена допустима только одна, активная и загруженная");
                }
                else
                {
                    ValidateAssetPath(scene.path);
                    if (!scene.path.EndsWith(".unity", StringComparison.OrdinalIgnoreCase) ||
                        !File.Exists(Path.Combine(Root, scene.path)) || !paths.Add(scene.path))
                        throw new InvalidOperationException("Недопустимая/отсутствующая сцена: " + scene.path);
                }
                if (scene.is_loaded) loaded++;
                if (scene.is_active) { if (!scene.is_loaded) throw new InvalidOperationException("Активная сцена не загружена"); active++; }
            }
            if (setup.scenes.Length != 0 && (loaded == 0 || active != 1))
                throw new InvalidOperationException("Нужна ровно одна активная загруженная сцена");
            string prefab = setup.prefab_path ?? setup.prefab_asset_path;
            if (!string.IsNullOrEmpty(prefab))
            {
                ValidateAssetPath(prefab);
                if (!prefab.EndsWith(".prefab", StringComparison.OrdinalIgnoreCase) || !File.Exists(Path.Combine(Root, prefab)))
                    throw new InvalidOperationException("Prefab отсутствует");
            }
        }

        private static void RestoreSetup(BridgeSetup setup)
        {
            if (PrefabStageUtility.GetCurrentPrefabStage() != null) StageUtility.GoToMainStage();
            if (setup.scenes.Length == 0 || setup.scenes.Length == 1 && string.IsNullOrEmpty(setup.scenes[0].path))
                EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            else EditorSceneManager.RestoreSceneManagerSetup(setup.scenes.Select(scene => new SceneSetup {
                path = scene.path, isLoaded = scene.is_loaded, isActive = scene.is_active }).ToArray());
            string prefab = setup.prefab_path ?? setup.prefab_asset_path;
            if (!string.IsNullOrEmpty(prefab)) PrefabStageUtility.OpenPrefab(prefab);
        }
    }
