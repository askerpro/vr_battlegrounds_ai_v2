// Только изолированный инфраструктурный стенд; в рабочий проект не устанавливается.
using System;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

[InitializeOnLoad]
public static class EditorBrokerSmoke
{
    [Serializable] private sealed class Config { public string state_dir = null; }
    [Serializable] private sealed class Command { public string id = null; public string operation = null; public string path = null; }
    [Serializable] private sealed class Result
    {
        public string id;
        public string error;
        public string[] scenes;
        public string active;
        public string prefab;
        public int reload_count;
        public bool probe_loaded;
    }
    private const string ReloadKey = "EditorBrokerSmoke.ReloadCount";
    private static string FixtureDirectory;
    private static double NextPoll;

    static EditorBrokerSmoke()
    {
        SessionState.SetInt(ReloadKey, SessionState.GetInt(ReloadKey, 0) + 1);
        EditorApplication.update += Tick;
    }

    private static string DirectoryPath()
    {
        if (FixtureDirectory != null) return FixtureDirectory;
        string root = Path.GetFullPath(Path.Combine(Application.dataPath, ".."));
        var config = JsonUtility.FromJson<Config>(File.ReadAllText(Path.Combine(root,
            "Assets/Editor/VR_Battlegrounds/Debug/EditorBrokerLocal/bridge-config.json")));
        FixtureDirectory = Path.Combine(config.state_dir, "smoke-fixture");
        Directory.CreateDirectory(FixtureDirectory);
        return FixtureDirectory;
    }

    private static void Write(string path, object value)
    {
        string temporary = path + ".tmp";
        File.WriteAllText(temporary, JsonUtility.ToJson(value, true));
        if (File.Exists(path)) File.Delete(path);
        File.Move(temporary, path);
    }

    public static void Start()
    {
        try
        {
            foreach (string folder in new[] { "Scenes", "Output", "Foreign", "Prefabs" })
                if (!AssetDatabase.IsValidFolder("Assets/" + folder)) AssetDatabase.CreateFolder("Assets", folder);
            Scene first = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            new GameObject("FirstSceneObject");
            EditorSceneManager.SaveScene(first, "Assets/Scenes/A.unity");
            Scene second = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Additive);
            SceneManager.SetActiveScene(second);
            new GameObject("SecondSceneObject");
            EditorSceneManager.SaveScene(second, "Assets/Scenes/B.unity");
            SceneManager.SetActiveScene(first);
            var prefab = new GameObject("FixturePrefab");
            PrefabUtility.SaveAsPrefabAsset(prefab, "Assets/Prefabs/Fixture.prefab");
            UnityEngine.Object.DestroyImmediate(prefab);
            EditorSceneManager.SaveScene(first);
            foreach (string path in new[] { "Assets/Output/Generated.mat", "Assets/Foreign/Foreign.mat" })
            {
                var material = new Material(Shader.Find("Hidden/InternalErrorShader"));
                AssetDatabase.CreateAsset(material, path);
                AssetDatabase.SaveAssetIfDirty(material);
            }
            Write(Path.Combine(DirectoryPath(), "ready.json"), new Result { error = "" });
        }
        catch (Exception exception)
        {
            Write(Path.Combine(DirectoryPath(), "ready.json"), new Result { error = exception.ToString() });
        }
    }

    private static void Tick()
    {
        if (EditorApplication.timeSinceStartup < NextPoll) return;
        NextPoll = EditorApplication.timeSinceStartup + .1;
        try
        {
            string directory = DirectoryPath();
            string path = Path.Combine(directory, "command.json");
            if (!File.Exists(path)) return;
            Command command = JsonUtility.FromJson<Command>(File.ReadAllText(path));
            string responsePath = Path.Combine(directory, command.id + ".json");
            if (File.Exists(responsePath)) return;
            var result = new Result { id = command.id, error = "" };
            try
            {
                switch (command.operation)
                {
                    case "dirty_asset":
                        var material = AssetDatabase.LoadAssetAtPath<Material>(command.path);
                        material.name = material.name + "Changed";
                        EditorUtility.SetDirty(material);
                        break;
                    case "generate_asset":
                        var generated = new Material(Shader.Find("Hidden/InternalErrorShader"));
                        AssetDatabase.CreateAsset(generated, "Assets/Output/RuntimeGenerated.mat");
                        generated.name = "DirtyGeneratedMaterial";
                        EditorUtility.SetDirty(generated);
                        break;
                    case "dirty_scene":
                        new GameObject("UnsavedFixtureObject");
                        EditorSceneManager.MarkSceneDirty(SceneManager.GetActiveScene());
                        break;
                    case "clean":
                        foreach (string asset in new[] { "Assets/Output/Generated.mat", "Assets/Foreign/Foreign.mat" })
                            AssetDatabase.SaveAssetIfDirty(AssetDatabase.LoadMainAssetAtPath(asset));
                        for (int index = 0; index < SceneManager.sceneCount; index++)
                        {
                            Scene scene = SceneManager.GetSceneAt(index);
                            if (scene.isDirty && scene.path.Length > 0) EditorSceneManager.SaveScene(scene);
                        }
                        break;
                    case "prefab": PrefabStageUtility.OpenPrefab("Assets/Prefabs/Fixture.prefab"); break;
                    case "close_prefab": StageUtility.GoToMainStage(); break;
                    case "play": EditorApplication.isPlaying = true; break;
                    case "stop": EditorApplication.isPlaying = false; break;
                    case "inspect": break;
                    case "quit": break;
                    default: throw new InvalidOperationException("Неизвестная команда стенда");
                }
                result.scenes = Enumerable.Range(0, SceneManager.sceneCount).Select(index => SceneManager.GetSceneAt(index).path).ToArray();
                result.active = SceneManager.GetActiveScene().path;
                PrefabStage prefab = PrefabStageUtility.GetCurrentPrefabStage();
                result.prefab = prefab == null ? "" : prefab.assetPath;
                result.reload_count = SessionState.GetInt(ReloadKey, 0);
                result.probe_loaded = AppDomain.CurrentDomain.GetAssemblies().Any(assembly => assembly.GetType("EditorBrokerReloadProbe", false) != null);
            }
            catch (Exception exception) { result.error = exception.ToString(); }
            Write(responsePath, result);
            if (command.operation == "quit") EditorApplication.delayCall += () => EditorApplication.Exit(0);
        }
        catch { /* Ошибка стенда будет видна таймаутом, без записи в игровой лог. */ }
    }
}
