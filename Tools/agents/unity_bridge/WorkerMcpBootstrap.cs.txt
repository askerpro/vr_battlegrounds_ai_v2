// Скопировать в локальный MCP package worker: подключение без изменения общих EditorPrefs.
using System;
using System.IO;
using System.Reflection;
using System.Threading.Tasks;
using MCPForUnity.Editor.Services;
using MCPForUnity.Editor.Services.Transport;
using UnityEditor;
using UnityEngine;

namespace MCPForUnity.Editor
{
    [InitializeOnLoad]
    public static class WorkerMcpBootstrap
    {
        [Serializable] private sealed class BridgeConfig
        {
            public string project_root = null;
            public string state_dir = null;
        }
        [Serializable] private sealed class Registration { public string editor_root = null; }
        [Serializable] private sealed class Result
        {
            public bool passed;
            public string project_root;
            public string error;
        }
        private static Task<bool> connect;
        private static string state;
        private static string root;

        static WorkerMcpBootstrap() { EditorApplication.update += Tick; }

        private static void Tick()
        {
            if (EditorApplication.isCompiling || EditorApplication.isUpdating) return;
            try
            {
                var reload = typeof(MCPServiceLocator).Assembly.GetType(
                    "MCPForUnity.Editor.Services.HttpBridgeReloadHandler", true);
                var resume = reload.GetProperty("IsResumePending",
                    BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic);
                if (resume == null) throw new InvalidOperationException("Неизвестный MCP reload API");
                if ((bool)resume.GetValue(null)) return;
                if (root == null)
                {
                    root = Path.GetFullPath(Path.Combine(Application.dataPath, ".."));
                    string configPath = Path.Combine(Application.dataPath,
                        "Editor/VR_Battlegrounds/Debug/EditorBrokerLocal/bridge-config.json");
                    if (!File.Exists(configPath)) { EditorApplication.update -= Tick; return; }
                    var config = JsonUtility.FromJson<BridgeConfig>(File.ReadAllText(configPath));
                    if (!Same(root, config.project_root)) throw new InvalidOperationException("Чужой bridge root");
                    state = Path.GetFullPath(config.state_dir);
                    var registration = JsonUtility.FromJson<Registration>(
                        File.ReadAllText(Path.Combine(state, "config.json")));
                    if (!Same(root, registration.editor_root)) throw new InvalidOperationException("Чужой worker root");
                }
                var manager = MCPServiceLocator.TransportManager;
                if (manager.IsRunning(TransportMode.Http)) { Finish(true, null); return; }
                if (connect == null) { connect = manager.StartAsync(TransportMode.Http); return; }
                if (!connect.IsCompleted) return;
                Finish(connect.GetAwaiter().GetResult() && manager.IsRunning(TransportMode.Http), null);
            }
            catch (Exception error) { Finish(false, error.Message); }
        }

        private static bool Same(string left, string right) => string.Equals(
            Path.GetFullPath(left), Path.GetFullPath(right), StringComparison.OrdinalIgnoreCase);

        private static void Finish(bool passed, string error)
        {
            EditorApplication.update -= Tick;
            if (state == null) return;
            string path = Path.Combine(state, "unity", "mcp-bootstrap.json");
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            File.WriteAllText(path, JsonUtility.ToJson(new Result
                { passed = passed, project_root = root, error = error }, true));
        }
    }
}
