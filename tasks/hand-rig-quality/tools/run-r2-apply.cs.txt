        string overall = fit?.Overall ?? "missing", palm = fit?.PalmSeverity ?? "missing";
        string status = fit?.Status ?? "missing";
        fitStatus[status] = fitStatus.ContainsKey(status) ? fitStatus[status] + 1 : 1;
        fitOverall[overall] = fitOverall.ContainsKey(overall) ? fitOverall[overall] + 1 : 1;
        palmSeverity[palm] = palmSeverity.ContainsKey(palm) ? palmSeverity[palm] + 1 : 1;
        if (fit != null && fit.NormalOffsetRecommended) normalRecommended++;
        int curls = fit?.SuggestedCurlDeg?.Count ?? 0;
        if (curls != 0) curlRecommendedHands++;
        curlRecommendedFingers += curls;
        hands.Add(new { point.Pose, point.Weapon, point.GrabPoint, hand.Side, hand.Palm, hand.RuntimeVsWorldDeg, hand.MetacarpalInfo, fit });
    }
}
if (result == null || result.Points.Count != 17 || hands.Count != 34) failures.Add("Ожидались 17 точек / 34 руки.");
var reportedWritten = result == null ? new string[0] : result.Points.SelectMany(p => p.Written).Distinct().ToArray();
if (!allowed.SetEquals(reportedWritten)) failures.Add("Written не охватывает ровно 26 разрешённых ассетов.");
if (UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode || UnityEditor.EditorApplication.isCompiling || UnityEditor.EditorApplication.isUpdating) failures.Add("Редактор занят после пакета; дождаться завершения перед finish.");
string reportPath = System.IO.Path.Combine(request.OutDir, "r2-apply-validation.json");
var report = new { passed = failures.Count == 0, scope = "R2 initial authoring + static Fit; кандидат ревью, без contact corrections/gameplay/Quest/приёмки",
    status = result?.Status, operationError, extractionReport = result?.ReportPath, points = result?.Points.Count, handCount = hands.Count,
    allowed, reportedWritten, changed, actualWrittenPaths = changed.Where(p => allowed.Contains(p)).ToArray(), unexpected, newPoseFiles,
    before, after, foreignDirtyPaths, foreignDirtyBefore, foreignDirtyAfter, foreignDirtyChanges,
    guidsBefore, guidsAfter, guidChanges, identitiesBefore, identitiesAfter, identityChanges,
    nonGripBefore, nonGripAfter, nonGripChanges, assetIds, hands, fitStatus, fitOverall, palmSeverity, normalRecommended, curlRecommendedHands, curlRecommendedFingers,
    failures, warnings = result?.Warnings, prerequisites = prereqs, measurePath, measureSha };
System.IO.File.WriteAllText(reportPath, Newtonsoft.Json.JsonConvert.SerializeObject(report, Newtonsoft.Json.Formatting.Indented), new System.Text.UTF8Encoding(false));
progress("validation_saved", after.Count, paths.Count);
return new { report.passed, report.scope, report.status, report.points, report.handCount, changedCount = changed.Length,
    writtenCount = reportedWritten.Length, unexpectedCount = unexpected.Length, fitStatus, fitOverall, palmSeverity, normalRecommended, curlRecommendedHands, curlRecommendedFingers,
    foreignDirtyObjectCount = foreignDirtyBefore.Count, foreignDirtyPathCount = foreignDirtyPaths.Length, foreignDirtyChangeCount = foreignDirtyChanges.Count,
    failureCount = failures.Count, examples = failures.Concat(foreignDirtyChanges).Concat(nonGripChanges).Concat(identityChanges).Take(10).ToArray(), reportPath, report.extractionReport };
} catch (System.Exception inspectionError) {
    // Сохраняем известный исход и фактические байты даже при отказе последующего инспектора.
    var afterFailure = new System.Collections.Generic.Dictionary<string, string>();
    var inspectionErrors = new System.Collections.Generic.List<string> { inspectionError.ToString() };
    try { inspectForeignDirty(); }
    catch (System.Exception e) { inspectionErrors.Add("Foreign dirty inspection: " + e); }
    progress("failure_posthash", 0, paths.Count);
    foreach (string path in paths) {
        try { afterFailure[path] = System.IO.File.Exists(path) ? sha256(path) : null; }
        catch (System.Exception e) { inspectionErrors.Add(path + ": " + e.Message); }
        if (afterFailure.Count % 100 == 0) progress("failure_posthash", afterFailure.Count, paths.Count);
    }
    var changedFailure = afterFailure.Keys.Where(p => before[p] != afterFailure[p]).ToArray();
    string partialReportPath = System.IO.Path.Combine(request.OutDir, "r2-apply-validation.json");
    System.IO.File.WriteAllText(partialReportPath, Newtonsoft.Json.JsonConvert.SerializeObject(new { passed = false,
        scope = "R2 authoring/static Fit; инспекция неполная, приёмки нет", operationError, inspectionErrors,
        extractionReport = result?.ReportPath, status = result?.Status, reportedWritten = result?.Written,
        before, after = afterFailure, foreignDirtyPaths, foreignDirtyBefore, foreignDirtyAfter, foreignDirtyChanges,
        changed = changedFailure, actualWrittenPaths = changedFailure.Where(p => allowed.Contains(p)).ToArray(),
        unexpected = changedFailure.Where(p => !allowed.Contains(p)).ToArray(), failures, measurePath, measureSha }, Newtonsoft.Json.Formatting.Indented), new System.Text.UTF8Encoding(false));
    progress("failure_validation_saved", afterFailure.Count, paths.Count);
    return new { passed = false, failureCount = failures.Count + inspectionErrors.Count, examples = failures.Concat(inspectionErrors).Take(10).ToArray(), reportPath = partialReportPath, extractionReport = result?.ReportPath };
}
