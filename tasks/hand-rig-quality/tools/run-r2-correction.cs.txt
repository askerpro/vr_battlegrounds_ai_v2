System.Func<Newtonsoft.Json.Linq.JObject, string, System.Collections.Generic.List<Newtonsoft.Json.Linq.JToken>> grabTokensFor = (json, path) => {
    var payload = (json["MonoBehaviour"] ?? json) as Newtonsoft.Json.Linq.JObject;
    var primary = payload?["_grabPoint"] as Newtonsoft.Json.Linq.JObject;
    var additional = payload?["_additionalGrabPoints"] as Newtonsoft.Json.Linq.JArray;
    if (primary == null || additional == null || additional.Any(p => !(p is Newtonsoft.Json.Linq.JObject)))
        throw new System.InvalidOperationException("Неизвестная схема root UxrGrabbableObject JSON: " + path);
    var tokens = new System.Collections.Generic.List<Newtonsoft.Json.Linq.JToken> { primary };
    tokens.AddRange(additional);
    foreach (var point in selected.Where(p => p.Weapon == path)) {
        if (point.GrabPoint < 0 || point.GrabPoint >= tokens.Count || !(tokens[point.GrabPoint]["_avatarGripPoseEntries"] is Newtonsoft.Json.Linq.JArray))
            throw new System.InvalidOperationException("Нет выбранного grab index/avatar entries: " + path + "#" + point.GrabPoint);
        var entries = (Newtonsoft.Json.Linq.JArray)tokens[point.GrabPoint]["_avatarGripPoseEntries"];
        if (entries.Any(e => !(e is Newtonsoft.Json.Linq.JObject) || e["_avatarPrefabGuid"]?.Type != Newtonsoft.Json.Linq.JTokenType.String) ||
            entries.Count(e => (string)e["_avatarPrefabGuid"] == recordGuid) != 1)
            throw new System.InvalidOperationException("Целевая GUID запись должна быть единственной: " + path + "#" + point.GrabPoint);
    }
    return tokens;
};

        map[key + "|object"] = Newtonsoft.Json.JsonConvert.SerializeObject(new { t.name, t.gameObject.layer, t.gameObject.tag, t.gameObject.activeSelf,
            position = new[] { t.localPosition.x, t.localPosition.y, t.localPosition.z }, rotation = new[] { t.localRotation.x, t.localRotation.y, t.localRotation.z, t.localRotation.w }, scale = new[] { t.localScale.x, t.localScale.y, t.localScale.z } });
