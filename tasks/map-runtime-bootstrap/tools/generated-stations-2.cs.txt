System.Func<System.Collections.Generic.IEnumerable<System.Guid>, byte[]> snapshot = targets => {
    var body = new System.IO.MemoryStream(); var w = new System.IO.BinaryWriter(body);
    foreach (var id in targets)
    {
        var rs = new System.IO.MemoryStream(); var rw = new System.IO.BinaryWriter(rs);
        rw.Write(UltimateXR.Extensions.System.IO.BinaryWriterExt.UniqueComponentId); rw.Write(id.ToByteArray()); rw.Write(new byte[] { 1, 2, 3 }); rw.Flush();
        var record = rs.ToArray(); UltimateXR.Extensions.System.IO.BinaryWriterExt.WriteCompressedInt32(w, record.Length); w.Write(record);
    }
    w.Flush(); var b = body.ToArray(); var r = new byte[4 + b.Length];
    r[0] = (byte)UltimateXR.Core.Serialization.UxrSerializationFormat.BinaryUncompressed;
    System.BitConverter.GetBytes((ushort)UltimateXR.Core.UxrConstants.Serialization.CurrentBinaryVersion).CopyTo(r, 2);
    b.CopyTo(r, 4); return r; };

System.Func<VrBattlegrounds.Arsenal.ArsenalStationCompositionBinding, string, VrBattlegrounds.Maps.Runtime.MapRunKey, VrBattlegrounds.Maps.Runtime.MapRunConfig> serverConfig = (st, scene, key) => {
    var errors = new System.Collections.Generic.List<string>();
    var adapter = adapterType.GetMethod("Describe", BF).Invoke(null, new object[] { new[] { st }, preset, catalog, null, errors });
    if (errors.Count != 0) throw new System.InvalidOperationException("Describe: " + string.Join(", ", errors.ToArray()));

    var bindings = (VrBattlegrounds.Maps.Runtime.MapRunBindings)call(adapter, "Apply", new object[] { shell });
    var cat = new VrBattlegrounds.Maps.Runtime.MapRunResolverCatalog("warmup", new string[0], new[] {
        new VrBattlegrounds.Maps.Runtime.MapRunMapDescription(scene, VrBattlegrounds.Maps.Runtime.MapRunKind.Lobby, "content", 1, "arsenal", "arsenal-v1", new string[0]) });
    var res = VrBattlegrounds.Maps.Runtime.MapRunResolver.Resolve(new VrBattlegrounds.Maps.Runtime.MapRunRequest(key, scene, "", "content"), cat, bindings);
    if (!res.Passed) throw new System.InvalidOperationException("Resolve: " + string.Join(", ", System.Linq.Enumerable.ToArray(res.Errors)));
    return res.Config; };
